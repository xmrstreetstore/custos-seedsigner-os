start_file=start4x.elf
fixup_file=fixup4x.dat

# Kernel location
kernel=zImage

# How much memory in MB to assign to the GPU on Pi models having
gpu_mem=256

# Enable Camera module
start_x=1

# Enable SPI
dtparam=spi=on

# Set the bootloader delay to 0 seconds. The default is 1s if not specified
boot_delay=0
disable_splash=1

# raspberry firmware config to disable wifi and bluetooth 
# https://github.com/raspberrypi/firmware/blob/master/boot/overlays/README
dtoverlay=disable-wifi
dtoverlay=disable-bt
# The SEC1210 reader is a *serial* CCID device on /dev/ttyAMA0 (the PL011 UART).
# On Pi Zero W / Zero 2 W / Pi 3+ that UART belongs to Bluetooth by default, so
# disabling BT is required for the reader to work at all, not just hardening.

# bcm2835_mmc does not abusively poll when the MicroSD is removed.
dtoverlay=mmc
