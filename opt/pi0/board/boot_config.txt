start_file=start_x.elf
fixup_file=fixup_x.dat

# Kernel location
kernel=zImage

# How much memory in MB to assign to the GPU on Pi models having
gpu_mem=64

# Enable Camera module
start_x=1

# Enable SPI
dtparam=spi=on

# Set the bootloader delay to 0 seconds. The default is 1s if not specified
boot_delay=0
disable_splash=1
force_turbo=1
# The SEC1210 smartcard reader is a *serial* CCID device on /dev/ttyAMA0, which is
# the PL011 UART. On Pi Zero W, Zero 2 W and Pi 3+ that UART is wired to Bluetooth
# by default, so disabling BT is required for the reader to work at all - it is not
# only a hardening choice. Wifi goes too: this device has no business having either.
dtoverlay=disable-wifi
dtoverlay=disable-bt

# Freeing the UART is not enough on its own: the firmware must also initialise
# it and hold its clock steady, or the baud rate drifts and the reader answers
# erratically. force_turbo=1 above already pins the core clock, but state the
# requirement explicitly rather than depending on that side effect.
enable_uart=1

# The PN532 NFC reader is on I2C bus 1 (GPIO2 SDA, GPIO3 SCL), since the UART
# already belongs to the SEC1210. libnfc reaches it as pn532_i2c:/dev/i2c-1.
dtparam=i2c_arm=on

# Use bcm2835_mmc rather than sdhost, which does not abusively poll when the
# MicroSD is removed - and SeedSigner is normally run with the card out.
dtoverlay=mmc
